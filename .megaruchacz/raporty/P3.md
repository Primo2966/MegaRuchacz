Kopia robocza workera (izolacja worktree w Claude Code) odmawia KAZDEGO wywolania
`powershell` z narzedzia Bash - takze `powershell -Command "1+1"` - komunikatem
"cannot be shown not to run git". Innej drogi do PowerShella worker nie ma (brak
narzedzia PowerShell), a obchodzenie straznika przez `node child_process` byloby

Sprawdzone bylo tylko to, co da sie bez PowerShella:
- rownowaga nawiasow (), {}, [] we wszystkich trzech plikach - przed i po zmianie
  (zgrubny tokenizer w node, uwzglednia napisy, komentarze i here-stringi),
- BOM + CRLF w `nadzorca.ps1` i `stan-nadzorcy.ps1` zachowane; `koszt-pamieci.ps1`
  dalej czyste ASCII z CRLF,
- brak spacji po backtickach konczacych linie.

To jest dokladnie przypadek z CLAUDE.md ("twierdzenie o mechanice narzedzia"):
izolacja worktree blokuje tu nie tylko git poza kopia, ale cale PowerShellowe
testowanie - dla projektu pisanego w PowerShellu to istotna roznica.

## Tresc test-warstwy.ps1

```powershell
# Sprawdzenie zakladki "Warstwy pamieci" (zadanie P3). Uruchomienie:
#   powershell -ExecutionPolicy Bypass -File test-warstwy.ps1 -Kopia <katalog kopii z nowym kodem> [-Wzor C:\dev\claude-worker] [-BezOkna]
# Nic nie zapisuje w katalogu domowym ani w repo; okno testowe ma wlasny zamek
# (nie koliduje z dzialajacym nadzorca) i zamyka sie samo.
param(
  [Parameter(Mandatory = $true)][string]$Kopia,
  [string]$Wzor = "C:\dev\claude-worker",
  [switch]$BezOkna
)
$ErrorActionPreference = "Stop"
$tmp = Join-Path ([System.IO.Path]::GetTempPath()) ("test-warstwy-" + [guid]::NewGuid().ToString("N").Substring(0, 8))
New-Item -ItemType Directory -Force -Path $tmp | Out-Null
$wyniki = @()
function Wynik($nazwa, $ok, $opis) {
  $script:wyniki += [pscustomobject]@{ Test = $nazwa; OK = [bool]$ok; Opis = $opis }
  $z = "NIE"; if ($ok) { $z = "TAK" }
  Write-Host ("[{0}] {1} - {2}" -f $z, $nazwa, $opis)
}
function Wolaj($skrypt, [string[]]$argumenty) {
  $wy = & powershell.exe -NoProfile -NonInteractive -ExecutionPolicy Bypass -File $skrypt @argumenty 2>&1
  return [pscustomobject]@{ Kod = $LASTEXITCODE; Tekst = (($wy | ForEach-Object { "$_" }) -join "`n") }
}

$nowy = Join-Path $Kopia "narzedzia\koszt-pamieci.ps1"
$stary = Join-Path $Wzor "narzedzia\koszt-pamieci.ps1"

# 1a. stare tryby: ten sam wynik przed i po (oba z tym samym -Zrodlo)
$tryby = [ordered]@{
  "domyslny" = @("-Zwykly"); "zwiezle" = @("-Zwiezle"); "dane" = @("-Dane", "-Zwykly")
  "rozbicie" = @("-Rozbicie", "-Zwykly"); "sufity" = @("-TylkoSufity", "-Zwykly")
  "projekt"  = @("-Zwykly", "-Projekt", $Wzor)
}
foreach ($k in $tryby.Keys) {
  $a = @("-Zrodlo", $Wzor) + $tryby[$k]
  $s = Wolaj $stary $a
  $n = Wolaj $nowy $a
  # linia "wygenerowano"/godziny moga sie roznic o minute - porownujemy bez cyfr godzin
  $sT = $s.Tekst -replace '\d{2}:\d{2}(:\d{2})?', 'GG:MM'
  $nT = $n.Tekst -replace '\d{2}:\d{2}(:\d{2})?', 'GG:MM'
  Wynik "stary tryb $k" (($sT -eq $nT) -and ($s.Kod -eq $n.Kod)) "kod przed $($s.Kod), po $($n.Kod); tekst $(if ($sT -eq $nT) { 'identyczny' } else { 'ROZNY' })"
  if ($sT -ne $nT) {
    $sT | Set-Content (Join-Path $tmp "$k-przed.txt"); $nT | Set-Content (Join-Path $tmp "$k-po.txt")
    Write-Host "     roznica: fc $tmp\$k-przed.txt $tmp\$k-po.txt"
  }
}

# 1b. -Warstwy daje poprawny JSON
$r = Wolaj $nowy @("-Warstwy")
$j = $null
try { $j = $r.Tekst | ConvertFrom-Json; Wynik "JSON -Warstwy" ($null -ne $j.Warstwy) "kod $($r.Kod), warstw $(@($j.Warstwy).Count), uwag $(@($j.Uwagi).Count)" }
catch { Wynik "JSON -Warstwy" $false "ConvertFrom-Json: $($_.Exception.Message); poczatek: $($r.Tekst.Substring(0, [math]::Min(300, $r.Tekst.Length)))" }
if ($j) {
  $j.Warstwy | Format-Table Kiedy, Trwalosc, Stan, Znaki, Rodzaj, Nazwa -AutoSize | Out-String -Width 200 | Write-Host
  $bez = @($j.Warstwy | Where-Object { -not $_.Nazwa -or -not $_.Kiedy -or -not $_.Trwalosc -or -not $_.Stan })
  Wynik "komplet pol" ($bez.Count -eq 0) "warstw bez nazwy/kiedy/trwalosci/stanu: $($bez.Count)"
  Wynik "ASCII na wyjsciu" ($r.Tekst -notmatch '[^\x00-\x7F]') "polskie znaki jako \uXXXX"
  $biez = @($j.Warstwy | Where-Object { $_.Id -eq "claude-globalny-biezace" })
  Wynik "podwarstwa Biezace ma tresc" (($biez.Count -eq 1) -and ("$($biez[0].Tresc)" -match 'Bie')) "znakow: $($biez[0].Znaki)"
  $sesja = @($j.Warstwy | Where-Object { $_.Sciezka -like "*\mr\megaruchacz-sesja.json" })
  Wynik "sesja.json w nieuzywanych" (($sesja.Count -eq 1) -and ($sesja[0].Kiedy -eq "nieuzywane")) "$($sesja[0].Kiedy) - $($sesja[0].Opis)"
  $lore = @($j.Warstwy | Where-Object { $_.Id -eq "lore" })
  Wynik "baza Lore bez tresci" (($lore.Count -eq 1) -and ($null -eq $lore[0].Tresc) -and ($lore[0].Bajty -gt 0)) "bajtow $($lore[0].Bajty)"
}

# 2. proba negatywna: pusty katalog domowy - warstwy maja zostac na liscie jako "brak"
$pusty = Join-Path $tmp "pusty-dom"
New-Item -ItemType Directory -Force -Path $pusty | Out-Null
$r = Wolaj $nowy @("-Warstwy", "-KatalogDomowy", $pusty)
try {
  $jn = $r.Tekst | ConvertFrom-Json
  $gl = @($jn.Warstwy | Where-Object { $_.Id -eq "claude-globalny" })
  Wynik "negatywna: CLAUDE.md globalny" (($gl.Count -eq 1) -and ($gl[0].Stan -eq "brak") -and ($gl[0].Brak -like "nie ma pliku*")) "stan '$($gl[0].Stan)': $($gl[0].Brak)"
  $pod = @($jn.Warstwy | Where-Object { $_.Rodzic -eq "claude-globalny" })
  Wynik "negatywna: podwarstwy zostaja" (($pod.Count -ge 3) -and (@($pod | Where-Object { $_.Stan -ne "brak" }).Count -eq 0)) "$($pod.Count) podwarstw, wszystkie 'brak'"
  $inne = @($jn.Warstwy | Where-Object { $_.Id -in @("pamiec-natywna", "wiedza", "lore", "przypomnienie") })
  Wynik "negatywna: pozostale brakujace" (($inne.Count -eq 4) -and (@($inne | Where-Object { $_.Stan -ne "brak" }).Count -eq 0)) (($inne | ForEach-Object { "$($_.Id)=$($_.Stan)" }) -join ", ")
} catch { Wynik "negatywna: JSON" $false $_.Exception.Message }

# 3. okno: kopia nadzorcy z wlasnym zamkiem, bez dozoru, sama przelacza na warstwy,
#    klika kazda warstwe, robi zrzut i sie zamyka
if (-not $BezOkna) {
  $zrodloOkna = Get-Content -LiteralPath (Join-Path $Kopia "zasobnik\nadzorca.ps1") -Raw -Encoding UTF8
  $stanPlik = Join-Path $Kopia "zasobnik\stan-nadzorcy.ps1"
  $raportOkna = Join-Path $tmp "okno.txt"
  $zrzut = Join-Path $tmp "okno-warstwy.png"
  $test = @"
`$script:TestZegar = New-Object System.Windows.Forms.Timer
`$script:TestZegar.Interval = 12000
`$script:TestZegar.Add_Tick({
  `$script:TestZegar.Stop()
  `$l = @()
  try {
    `$script:BWarstwy.PerformClick()
    [System.Windows.Forms.Application]::DoEvents()
    `$l += "podsumowanie: " + `$script:LWarstwy.Text
    `$l += "pozycji na liscie: " + `$script:ListaWarstw.Items.Count + ", grup: " + `$script:ListaWarstw.Groups.Count
    foreach (`$it in `$script:ListaWarstw.Items) {
      `$it.Selected = `$true
      [System.Windows.Forms.Application]::DoEvents()
      `$p = `$script:PodgladWarstwy.Text
      `$l += ("{0} | {1} | {2} | grupa: {3} | podglad {4} zn.: {5}" -f `$it.Text.Trim(), `$it.SubItems[1].Text, `$it.SubItems[2].Text, `$it.Group.Header, `$p.Length, (`$p -split "``r``n")[0])
    }
    `$bmp = New-Object System.Drawing.Bitmap(`$script:Okno.Width, `$script:Okno.Height)
    `$script:Okno.DrawToBitmap(`$bmp, (New-Object System.Drawing.Rectangle(0, 0, `$bmp.Width, `$bmp.Height)))
    `$bmp.Save("$zrzut")
    `$l += "zrzut: $zrzut"
    `$l += "przelacznik: " + `$script:BPrzeglad.Parent.Bounds + "; podtytul: " + `$script:LPodtytul.Bounds
  } catch { `$l += "WYWROTKA TESTU: " + `$_.Exception.Message + " @ " + `$_.InvocationInfo.PositionMessage }
  `$l += "wywrotki nadzorcy: " + (`$script:NadzWywrotki -join " || ")
  [System.IO.File]::WriteAllLines("$raportOkna", [string[]]`$l, (New-Object System.Text.UTF8Encoding(`$true)))
  `$script:Okno.Close()
  `$script:Ikona.Visible = `$false
  [System.Windows.Forms.Application]::Exit()
})
`$script:TestZegar.Start()
"@
  $kod = $zrodloOkna.Replace('. (Join-Path $PSScriptRoot "stan-nadzorcy.ps1")', ". `"$stanPlik`"")
  $kod = $kod.Replace('"Local\MegaRuchacz-Nadzorca"', '"Local\MegaRuchacz-Nadzorca-TEST-P3"')
  $kod = $kod.Replace('$script:Zegar.Start()', '# dozor wylaczony w tescie')
  $kod = $kod.Replace('if ($Pokaz) { Pokaz-Okno }', "if (`$Pokaz) { Pokaz-Okno }`r`n$test")
  foreach ($kotwica in @('MegaRuchacz-Nadzorca-TEST-P3', 'dozor wylaczony w tescie', 'TestZegar', $stanPlik)) {
    if (-not $kod.Contains($kotwica)) { Wynik "okno: przygotowanie" $false "nie podmienilem kotwicy '$kotwica'"; $BezOkna = $true }
  }
  if (-not $BezOkna) {
    $kopiaOkna = Join-Path $tmp "nadzorca-test.ps1"
    [System.IO.File]::WriteAllText($kopiaOkna, $kod, (New-Object System.Text.UTF8Encoding($true)))
    $r = Wolaj $kopiaOkna @("-Zrodlo", $Kopia, "-Pokaz", "-Proba")
    if (Test-Path $raportOkna) {
      $txt = Get-Content -LiteralPath $raportOkna -Encoding UTF8
      $txt | Write-Host
      $zle = @($txt | Where-Object { $_ -like "WYWROTKA*" -or $_ -match "NIE UDA" })
      Wynik "okno z zakladka" (($zle.Count -eq 0) -and (Test-Path $zrzut)) "kod $($r.Kod); zrzut $zrzut"
    } else {
      Wynik "okno z zakladka" $false "okno nie zostawilo raportu (kod $($r.Kod)): $($r.Tekst)"
    }
  }
}

Write-Host ""
Write-Host "Wynik: $(@($wyniki | Where-Object { $_.OK }).Count) z $($wyniki.Count) TAK. Pliki robocze: $tmp"
if (@($wyniki | Where-Object { -not $_.OK }).Count -gt 0) { exit 1 }
exit 0
```
